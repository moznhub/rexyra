## 5. Active expiry and the table size limit
z: HELLO 3 @@ proto id
z: CLIENT TRACKING ON
w: SET ex1 v PX 100
z: GET ex1
~sleep 400
~drain
w: CONFIG GET tracking-table-max-keys
z: MGET t1 t2 t3 t4 t5
w: INFO stats @@ tracking_total_keys tracking_total_items tracking_total_prefixes
w: CONFIG SET tracking-table-max-keys 2
~sleep 300
~drain
w: INFO stats @@ tracking_total_keys tracking_total_items tracking_total_prefixes
w: CONFIG SET tracking-table-max-keys 0
z: MGET u1 u2 u3 u4 u5
~sleep 300
~drain
w: INFO stats @@ tracking_total_keys tracking_total_items tracking_total_prefixes
w: CONFIG SET tracking-table-max-keys -1
w: CONFIG SET tracking-table-max-keys 1000000
